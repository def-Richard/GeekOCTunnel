cmake_minimum_required(VERSION 3.22)
project(GeekOCTunnelDesktop VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network)
option(BUILD_LEGACY_WIDGETS "Build the original P0 configuration window" OFF)
include(CTest)

add_library(desktop-config STATIC core/ProfileStore.cpp)
target_include_directories(desktop-config PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(desktop-config PUBLIC Qt6::Core Qt6::Network)

add_library(desktop-bridge STATIC core/DesktopBridge.cpp)
target_link_libraries(desktop-bridge PUBLIC desktop-config)
add_executable(geekoctunnel-bridge bridge/main.cpp)
target_link_libraries(geekoctunnel-bridge PRIVATE desktop-bridge)

if(BUILD_LEGACY_WIDGETS)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    qt_add_executable(geekoctunnel-desktop ui/main.cpp)
    target_link_libraries(geekoctunnel-desktop PRIVATE desktop-config Qt6::Widgets)
    set_target_properties(geekoctunnel-desktop PROPERTIES WIN32_EXECUTABLE TRUE)
endif()

# The probe uses the actual public header, never a handwritten ABI declaration.
find_path(OPENCONNECT_INCLUDE_DIR openconnect.h
    HINTS ${CMAKE_CURRENT_SOURCE_DIR}/../native/src/openconnect)
if(NOT OPENCONNECT_INCLUDE_DIR)
    message(FATAL_ERROR "Install libopenconnect-dev or set OPENCONNECT_INCLUDE_DIR to the pinned source directory")
endif()
add_executable(geekoctunnel-probe probes/openconnect_probe.cpp)
target_include_directories(geekoctunnel-probe PRIVATE ${OPENCONNECT_INCLUDE_DIR})
target_link_libraries(geekoctunnel-probe PRIVATE Qt6::Core Qt6::Network)
if(WIN32)
    target_compile_definitions(geekoctunnel-probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(geekoctunnel-probe PRIVATE ws2_32 advapi32)
    add_executable(geekoctunnel-wintun-probe probes/wintun_probe.cpp)
    target_include_directories(geekoctunnel-wintun-probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${OPENCONNECT_INCLUDE_DIR})
    target_compile_definitions(geekoctunnel-wintun-probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(geekoctunnel-wintun-probe PRIVATE Qt6::Core iphlpapi ws2_32)
    add_executable(geekoctunnel-helper helper/windows_main.cpp)
    set_target_properties(geekoctunnel-helper PROPERTIES AUTOMOC OFF MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_include_directories(geekoctunnel-helper PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${OPENCONNECT_INCLUDE_DIR})
    target_compile_definitions(geekoctunnel-helper PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    target_compile_options(geekoctunnel-helper PRIVATE /W4 /utf-8 /permissive-)
    target_link_libraries(geekoctunnel-helper PRIVATE iphlpapi ws2_32 advapi32 bcrypt)
    add_executable(geekoctunnel-core core/SessionMain.cpp)
    target_include_directories(geekoctunnel-core PRIVATE ${OPENCONNECT_INCLUDE_DIR})
    target_compile_definitions(geekoctunnel-core PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX QT_NO_KEYWORDS)
    target_compile_options(geekoctunnel-core PRIVATE /W4 /utf-8 /permissive-)
    target_link_libraries(geekoctunnel-core PRIVATE desktop-config Qt6::Core Qt6::Network ws2_32 advapi32 shell32)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(geekoctunnel-core core/SessionMain.cpp)
    target_include_directories(geekoctunnel-core PRIVATE ${OPENCONNECT_INCLUDE_DIR})
    target_compile_definitions(geekoctunnel-core PRIVATE QT_NO_KEYWORDS _GNU_SOURCE)
    target_compile_options(geekoctunnel-core PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(geekoctunnel-core PRIVATE desktop-config Qt6::Core Qt6::Network)
    add_executable(geekoctunnel-helper helper/linux_main.cpp)
    set_target_properties(geekoctunnel-helper PROPERTIES AUTOMOC OFF)
    target_include_directories(geekoctunnel-helper PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(geekoctunnel-helper PRIVATE QT_NO_KEYWORDS _GNU_SOURCE)
    target_compile_options(geekoctunnel-helper PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(geekoctunnel-helper PRIVATE Qt6::Core Qt6::Network)
endif()

foreach(target desktop-config desktop-bridge geekoctunnel-bridge geekoctunnel-desktop geekoctunnel-probe)
    if(NOT TARGET ${target})
        continue()
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(desktop-contract-tests tests/contracts_test.cpp)
    target_link_libraries(desktop-contract-tests PRIVATE desktop-config Qt6::Test)
    target_compile_definitions(desktop-contract-tests PRIVATE
        FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../contracts/fixtures/server-addresses.tsv")
    add_test(NAME desktop-contracts COMMAND desktop-contract-tests)
    add_executable(desktop-bridge-tests tests/bridge_test.cpp)
    target_link_libraries(desktop-bridge-tests PRIVATE desktop-bridge Qt6::Test)
    add_test(NAME desktop-bridge COMMAND desktop-bridge-tests)
    add_executable(desktop-network-plan-tests tests/network_plan_test.cpp)
    target_include_directories(desktop-network-plan-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(desktop-network-plan-tests PRIVATE Qt6::Core Qt6::Network Qt6::Test)
    add_test(NAME desktop-network-plan COMMAND desktop-network-plan-tests)
    if(WIN32)
        add_executable(desktop-windows-path-tests tests/windows_paths_test.cpp)
        target_include_directories(desktop-windows-path-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        target_compile_definitions(desktop-windows-path-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_link_libraries(desktop-windows-path-tests PRIVATE Qt6::Core Qt6::Test ws2_32)
        add_test(NAME desktop-windows-paths COMMAND desktop-windows-path-tests)
    endif()
    if(BUILD_LEGACY_WIDGETS)
        add_test(NAME desktop-ui-smoke COMMAND geekoctunnel-desktop --smoke-test)
        set_tests_properties(desktop-ui-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
    endif()
endif()

install(TARGETS geekoctunnel-bridge geekoctunnel-probe RUNTIME DESTINATION bin)
